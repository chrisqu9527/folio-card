```
2x2 grid, 16:9, do this for 4 non-copyrighted folklore stories parents tell their children:  INPUT = {
    "topic": [ANY_TOPIC],
    "topic_set": [OPTIONAL],
    "panel_count": [OPTIONAL],
    "visual_context": [OPTIONAL]
}

class UniversalConceptualMiniatureWorldCompiler:

    HARD_CODING = False
    INFERENCE_MODE = True

    def diagnose_topic(self, topic):
        return {
            "identity": infer_topic_identity(topic),
            "domain": infer_subject_domain(topic),
            "defining_traits": infer_distinctive_attributes(topic),
            "symbolism": infer_meaningful_visual_associations(topic),
            "context": infer_historical_cultural_or_functional_context(topic)
        }

    def generate_subject_set(self, topic):
        return infer_comparable_topics_or_subtopics(
            topic,
            count=INPUT["panel_count"],
            prioritize_visual_diversity=True,
            preserve_unifying_theme=True
        )

    def design_metaphor(self, subject):
        return {
            "physical_object":
                infer_strongest_metaphorical_artifact(subject),

            "conceptual_connection":
                explain_object_to_subject_relationship(subject),

            "material":
                infer_semantically_relevant_material(subject),

            "scale":
                infer_monumental_object_scale(subject),

            "form":
                preserve_recognizable_symbolic_geometry(subject),

            "color":
                infer_contextual_accent_palette(subject)
        }

    def generate_miniature_world(self, subject, metaphor):
        return {
            "environment":
                infer_topic_relevant_environment(subject),

            "architecture":
                infer_contextual_structures_or_none(subject),

            "figures":
                infer_relevant_miniature_agents(subject),

            "activities":
                infer_meaningful_actions(subject, metaphor),

            "secondary_objects":
                infer_contextual_artifacts(subject),

            "connections":
                integrate_illustrations_with_metaphor(metaphor)
        }

    def compose_panel(self, subject):
        metaphor = self. design_metaphor(subject)

        world = self.generate_miniature_world(
            subject,
            metaphor
        )

        return render_mixed_media_diorama(
            hero=photorealistic_symbolic_object(metaphor),
            surroundings=fine_monochrome_line_art(world),
            background="clean editorial ivory",
            title=generate_readable_subject_label(subject),
            perspective=infer_best_illustrative_camera(subject),
            whitespace=optimize_negative_space()
        )

    def render_final(self):
        subjects = self.generate_subject_set(INPUT["topic"])

        panels = [
            self.compose_panel(self.diagnose_topic(subject))
            for subject in subjects
        ]

        return render_adaptive_editorial_grid(
            panels=panels,
            layout=infer_comparison_layout(panels),
            constraints=[
                "no hard-coded object",
                "no fixed civilizations",
                "no fixed landmarks",
                "no generic miniature activities",
                "every metaphor must communicate its subject",
                "3D objects contrast with fine 2D line art",
                "premium conceptual encyclopedia quality"
            ]
        )
```
